--==================== UI HELPERS ====================

local function addCorner(object, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius)
    corner.Parent = object
end

local function addStroke(object, color, thickness)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color
    stroke.Thickness = thickness
    stroke.Parent = object
end

local function makeLabel(parent, name, text, size, position, textSize, color)
    local label = Instance.new("TextLabel")
    label.Name = name
    label.Size = size
    label.Position = position
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = color or Color3.fromRGB(240, 240, 248)
    label.TextSize = textSize
    label.Font = Enum.Font.GothamMedium
    label.TextWrapped = true
    label.ZIndex = 5
    label.Parent = parent
    return label
end

local function makeButton(parent, name, text, size, position, color)
    local button = Instance.new("TextButton")
    button.Name = name
    button.Size = size
    button.Position = position
    button.BackgroundColor3 = color
    button.Text = text
    button.TextColor3 = Color3.fromRGB(255, 255, 255)

    button.AutoButtonColor = true
    button.Active = true
    button.ZIndex = 7

    return button
end

local function makePanel(name, height, strokeColor)
    local panel = Instance.new("Frame")
    panel.Name = name
    panel.AnchorPoint = Vector2.new(0.5, 0.5)
    panel.Position = UDim2.fromScale(0.5, 0.5)
    panel.Size = UDim2.new(0.88, 0, 0, height)
    panel.BackgroundColor3 = Color3.fromRGB(15, 15, 23)
    panel.BorderSizePixel = 0
    panel.Active = true
    panel.Visible = false
    panel.ZIndex = 3
    panel.Parent = overlay

    addCorner(panel, 16)
    addStroke(panel, strokeColor, 1.5)

    local constraint = Instance.new("UISizeConstraint")
    constraint.MinSize = Vector2.new(260, height)
    constraint.MaxSize = Vector2.new(440, height)
    constraint.Parent = panel

    return panel
end

--==================== EXPIRY / WORKSPACE MARKER ====================

local startTime = os.time()
local expiresAt = startTime + DURATION

local function createExpiryMarker()
    -- Xóa marker cũ nếu có
    local oldMarker = Workspace:FindFirstChild("Roblox_Band")
    if oldMarker then
        oldMarker:Destroy()
    end

    -- Đối tượng trong Workspace, không phải tệp trên ổ đĩa
    local marker = Instance.new("Folder")
    marker.Name = "Roblox_Band"

    local status = Instance.new("StringValue")
    status.Name = "Status"
    status.Value = "Ban test duration completed"
    status.Parent = marker

    local completedAt = Instance.new("NumberValue")
    completedAt.Name = "CompletedAtUnix"
    completedAt.Value = os.time()
    completedAt.Parent = marker

    local playerName = Instance.new("StringValue")
    playerName.Name = "PlayerName"
    playerName.Value = Player.Name
    playerName.Parent = marker

    marker.Parent = Workspace
end

task.spawn(function()
    while gui.Parent do
        local remaining = math.max(0, expiresAt - os.time())

        local hours = math.floor(remaining / 3600)
        local minutes = math.floor((remaining % 3600) / 60)
        local seconds = remaining % 60

        countdown.Text = string.format(
            "%02d:%02d:%02d",
            hours, minutes, seconds
        )

        if remaining <= 0 then
            break
        end

        task.wait(1)
    end

    if not gui.Parent then
        return
    end

    -- Chỉ tạo marker sau khi đủ 8 giờ
    createExpiryMarker()

    gui:Destroy()

    if blur and blur.Parent then
        blur:Destroy()
    end
end)
