local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- Xóa giao diện cũ nếu đã tồn tại
local oldGui = playerGui:FindFirstChild("AccountWarningUI")
if oldGui then
    oldGui:Destroy()
end

-- ScreenGui
local gui = Instance.new("ScreenGui")
gui.Name = "AccountWarningUI"
gui.ResetOnSpawn = false
gui.IgnoreGuiInset = true
gui.DisplayOrder = 999999
gui.Parent = playerGui

-- Nền tối
local overlay = Instance.new("Frame")
overlay.Name = "Overlay"
overlay.Size = UDim2.fromScale(1, 1)
overlay.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
overlay.BackgroundTransparency = 0.35
overlay.BorderSizePixel = 0
overlay.Parent = gui

-- Bảng cảnh cáo
local panel = Instance.new("Frame")
panel.Name = "WarningPanel"
panel.AnchorPoint = Vector2.new(0.5, 0.5)
panel.Position = UDim2.fromScale(0.5, 0.5)
panel.Size = UDim2.new(0.88, 0, 0, 290)
panel.BackgroundColor3 = Color3.fromRGB(18, 18, 23)
panel.BorderSizePixel = 0
panel.Parent = overlay

local sizeLimit = Instance.new("UISizeConstraint")
sizeLimit.MaxSize = Vector2.new(420, 290)
sizeLimit.MinSize = Vector2.new(260, 290)
sizeLimit.Parent = panel

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 16)
corner.Parent = panel

local stroke = Instance.new("UIStroke")
stroke.Color = Color3.fromRGB(255, 55, 65)
stroke.Thickness = 1.5
stroke.Transparency = 0.1
stroke.Parent = panel

-- Thanh đỏ phía trên
local topBar = Instance.new("Frame")
topBar.Size = UDim2.new(1, 0, 0, 5)
topBar.BackgroundColor3 = Color3.fromRGB(255, 45, 55)
topBar.BorderSizePixel = 0
topBar.Parent = panel

local barCorner = Instance.new("UICorner")
barCorner.CornerRadius = UDim.new(0, 8)
barCorner.Parent = topBar

-- Biểu tượng cảnh báo
local icon = Instance.new("TextLabel")
icon.Size = UDim2.new(1, 0, 0, 55)
icon.Position = UDim2.new(0, 0, 0, 17)
icon.BackgroundTransparency = 1
icon.Text = "⚠"
icon.TextColor3 = Color3.fromRGB(255, 65, 75)
icon.TextSize = 42
icon.Font = Enum.Font.GothamBold
icon.Parent = panel

-- Tiêu đề
local title = Instance.new("TextLabel")
title.Position = UDim2.new(0, 15, 0, 75)
title.Size = UDim2.new(1, -30, 0, 35)
title.BackgroundTransparency = 1
title.Text = "CẢNH CÁO TÀI KHOẢN"
title.TextColor3 = Color3.fromRGB(255, 65, 75)
title.TextSize = 20
title.Font = Enum.Font.GothamBold
title.Parent = panel

-- Nội dung
local message = Instance.new("TextLabel")
message.Position = UDim2.new(0, 22, 0, 120)
message.Size = UDim2.new(1, -44, 0, 75)
message.BackgroundTransparency = 1
message.Text = "Lần này là cảnh cáo!\nNếu tiếp tục tái phạm,\ntài khoản sẽ bị BAN vĩnh viễn."
message.TextColor3 = Color3.fromRGB(235, 235, 240)
message.TextSize = 15
message.Font = Enum.Font.Gotham
message.TextWrapped = true
message.TextXAlignment = Enum.TextXAlignment.Center
message.TextYAlignment = Enum.TextYAlignment.Center
message.Parent = panel

-- Nút mở khóa tài khoản
local button = Instance.new("TextButton")
button.Name = "UnlockAccountButton"
button.Position = UDim2.new(0, 22, 1, -62)
button.Size = UDim2.new(1, -44, 0, 42)
button.BackgroundColor3 = Color3.fromRGB(220, 40, 50)
button.BorderSizePixel = 0
button.Text = "MỞ KHÓA TÀI KHOẢN"
button.TextColor3 = Color3.fromRGB(255, 255, 255)
button.TextSize = 14
button.Font = Enum.Font.GothamBold
button.AutoButtonColor = true
button.Parent = panel

local buttonCorner = Instance.new("UICorner")
buttonCorner.CornerRadius = UDim.new(0, 9)
buttonCorner.Parent = button

-- Hiệu ứng xuất hiện
panel.Size = UDim2.new(0.82, 0, 0, 270)

TweenService:Create(
    panel,
    TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
    {Size = UDim2.new(0.88, 0, 0, 290)}
):Play()

-- Nhấn nút để đóng bảng
local clicked = false

button.Activated:Connect(function()
    if clicked then
        return
    end

    clicked = true
    button.Text = "ĐANG ĐÓNG..."

    local fadeInfo = TweenInfo.new(
        0.25,
        Enum.EasingStyle.Quad,
        Enum.EasingDirection.Out
    )

    TweenService:Create(panel, fadeInfo, {
        BackgroundTransparency = 1,
        Position = UDim2.fromScale(0.5, 0.53)
    }):Play()

    TweenService:Create(overlay, fadeInfo, {
        BackgroundTransparency = 1
    }):Play()

    for _, object in ipairs(panel:GetDescendants()) do
        if object:IsA("TextLabel") or object:IsA("TextButton") then
            TweenService:Create(object, fadeInfo, {
                TextTransparency = 1,
                BackgroundTransparency = 1
            }):Play()
        elseif object:IsA("UIStroke") then
            TweenService:Create(object, fadeInfo, {
                Transparency = 1
            }):Play()
        elseif object:IsA("Frame") then
            TweenService:Create(object, fadeInfo, {
                BackgroundTransparency = 1
            }):Play()
        end
    end

    task.wait(0.3)
    gui:Destroy()
end)
