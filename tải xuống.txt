

--==================================================
-- NHAT HUB | PROXIMITY BLOCKER V3
-- CLIENT ONLY | DELTA ANDROID
--==================================================

local Workspace = game:GetService("Workspace")

local TAG = "NHAT_HUB_BLOCKED"

local running = true
local blockedCount = 0
local disabledPrompts = {}
local promptConnections = {}
local connections = {}

--==================================================
-- VÔ HIỆU HÓA PROMPT
--==================================================

local function disablePrompt(prompt)
    if not running or not prompt:IsA("ProximityPrompt") then
        return
    end

    pcall(function()
        prompt.Enabled = false
        prompt:SetAttribute(TAG, true)
    end)
end

--==================================================
-- XỬ LÝ PROMPT, TRÁNH KẾT NỐI TRÙNG
--==================================================

local function processPrompt(prompt)
    if not running
        or not prompt:IsA("ProximityPrompt")
        or promptConnections[prompt] then
        return
    end

    disabledPrompts[prompt] = prompt.Enabled
    disablePrompt(prompt)
    blockedCount += 1

    print("[NHAT HUB] Đã phát hiện:", prompt:GetFullName())

    promptConnections[prompt] =
        prompt:GetPropertyChangedSignal("Enabled"):Connect(function()
            if running and prompt.Parent and prompt.Enabled then
                disablePrompt(prompt)
                print("[NHAT HUB] Đã tắt lại:", prompt.Name)
            end
        end)

    -- Dọn kết nối khi prompt bị xóa
    table.insert(connections,
        prompt.AncestryChanged:Connect(function(_, parent)
            if parent == nil then
                local connection = promptConnections[prompt]
                if connection then
                    connection:Disconnect()
                    promptConnections[prompt] = nil
                end

                disabledPrompts[prompt] = nil
            end
        end)
    )
end

--==================================================
-- QUÉT PROMPT HIỆN CÓ
--==================================================

for _, obj in ipairs(Workspace:GetDescendants()) do
    processPrompt(obj)
end

--==================================================
-- THEO DÕI PROMPT MỚI
--==================================================

table.insert(connections, Workspace.DescendantAdded:Connect(function(obj)
    if obj:IsA("ProximityPrompt") then
        processPrompt(obj)
    end
end))

--==================================================
-- THÔNG TIN TRẠNG THÁI
--==================================================

print("================================")
print("[NHAT HUB] PROXIMITY BLOCKER V3")
print("Prompt đã xử lý:", blockedCount)
print("Trạng thái: ĐANG CHẠY")
print("Chế độ: Theo dõi liên tục")
print("================================")
