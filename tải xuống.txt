local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

--========================================================
-- ONE-TIME ACTIVATION GUARD
--========================================================

local ATTRIBUTE_NAME = "NHAT_WARNING_ALREADY_ACTIVATED"

-- Không kích hoạt lại nếu đã chạy trong phiên này
if player:GetAttribute(ATTRIBUTE_NAME) == true then
    return
end

-- Đánh dấu ngay để ngăn script chạy trùng
player:SetAttribute(ATTRIBUTE_NAME, true)

--========================================================
-- CLEANUP
--========================================================

local oldGui = playerGui:FindFirstChild("NHAT_AccountWarning")

if oldGui then
    oldGui:Destroy()
end

--========================================================
-- SCREEN GUI
--========================================================

local gui = Instance.new("ScreenGui")
gui.Name = "NHAT_AccountWarning"
gui.ResetOnSpawn = false
gui.IgnoreGuiInset = true
gui.DisplayOrder = 999999
gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
gui.Parent = playerGui

-- Nền tối
local overlay = Instance.new("Frame")
overlay.Name = "Overlay"
overlay.Size = UDim2.fromScale(1, 1)
overlay.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
overlay.BackgroundTransparency = 0.35
overlay.BorderSizePixel = 0
overlay.Parent = gui

--========================================================
-- MAIN PANEL
--========================================================

local panel = Instance.new("Frame")
panel.Name = "WarningPanel"
panel.AnchorPoint = Vector2.new(0.5, 0.5)
panel.Position = UDim2.fromScale(0.5, 0.5)
panel.Size = UDim2.new(0.88, 0, 0, 290)
panel.BackgroundColor3 = Color3.fromRGB(18, 18, 24)
panel.BorderSizePixel = 0
panel.ClipsDescendants = true
panel.Parent = overlay

local sizeConstraint = Instance.new("UISizeConstraint")
sizeConstraint.MinSize = Vector2.new(260, 290)
sizeConstraint.MaxSize = Vector2.new(420, 290)
sizeConstraint.Parent = panel

local panelCorner = Instance.new("UICorner")
panelCorner.CornerRadius = UDim.new(0, 16)
panelCorner.Parent = panel

local panelStroke = Instance.new("UIStroke")
panelStroke.Color = Color3.fromRGB(255, 55, 70)
panelStroke.Thickness = 1.5

panelStroke.Parent = panel

--========================================================
-- TOP ACCENT
--========================================================

local topBar = Instance.new("Frame")
topBar.Name = "TopBar"
topBar.Size = UDim2.new(1, 0, 0, 5)
topBar.BackgroundColor3 = Color3.fromRGB(255, 45, 60)
topBar.BorderSizePixel = 0
topBar.Parent = panel

--========================================================
-- WARNING ICON
--========================================================

local icon = Instance.new("TextLabel")
icon.Name = "WarningIcon"
icon.Position = UDim2.new(0, 0, 0, 17)
icon.Size = UDim2.new(1, 0, 0, 52)
icon.BackgroundTransparency = 1
icon.Text = "⚠"

icon.TextSize = 42
icon.Font = Enum.Font.GothamBold
icon.Parent = panel

--========================================================
-- TITLE
--========================================================

local title = Instance.new("TextLabel")
title.Name = "Title"
title.Position = UDim2.new(0, 15, 0, 75)

title.BackgroundTransparency = 1

title.TextSize = 19

title.Parent = panel

--========================================================
-- MESSAGE
--========================================================

local message = Instance.new("TextLabel")
message.Name = "WarningMessage"
message.Position = UDim2.new(0, 22, 0, 118)

--========================================================
-- UNLOCK BUTTON
--========================================================

local button = Instance.new("TextButton")
button.Name = "UnlockAccountButton"
button.Position = UDim2.new(0, 22, 1, -62)
button.Size = UDim2.new(1, -44, 0, 42)
button.BackgroundColor3 = Color3.fromRGB(220, 40, 52)

buttonCorner.CornerRadius = UDim.new(0, 9)
buttonCorner.Parent = button

--========================================================
-- INTRO ANIMATION
--========================================================

panel.Size = UDim2.new(0.78, 0, 0, 260)
panel.BackgroundTransparency = 0.15

TweenService:Create(
    panel,
    TweenInfo.new(
        0.4,
        Enum.EasingStyle.Back,
        Enum.EasingDirection.Out
    ),
    {
        Size = UDim2.new(0.88, 0, 0, 290),
        BackgroundTransparency = 0
    }
):Play()

--========================================================
-- CLOSE ANIMATION
--========================================================

local closing = false

button.Activated:Connect(function()
    if closing then
        return
    end

    closing = true
    button.Text = "ĐANG ĐÓNG..."

    local fadeInfo = TweenInfo.new(
        0.25,
        Enum.EasingStyle.Quad,
        Enum.EasingDirection.Out
    )

    local tweens = {}

    table.insert(tweens, TweenService:Create(
        panel,
        fadeInfo,
        {
            BackgroundTransparency = 1,
            Position = UDim2.fromScale(0.5, 0.53)
        }
    ))

    table.insert(tweens, TweenService:Create(
        overlay,
        fadeInfo,
        {
            BackgroundTransparency = 1
        }
    ))

    for _, object in ipairs(panel:GetDescendants()) do
        if object:IsA("TextLabel") or object:IsA("TextButton") then
            table.insert(tweens, TweenService:Create(
                object,
                fadeInfo,
                {
                    TextTransparency = 1,
                    BackgroundTransparency = 1
                }
            ))

        elseif object:IsA("UIStroke") then
            table.insert(tweens, TweenService:Create(
                object,
                fadeInfo,
                {
                    Transparency = 1
                }
            ))

        elseif object:IsA("Frame") then
            table.insert(tweens, TweenService:Create(
                object,
                fadeInfo,
                {
                    BackgroundTransparency = 1
                }
            ))
        end
    end

    for _, tween in ipairs(tweens) do
        tween:Play()
    end

    task.wait(0.3)

    if gui and gui.Parent then
        gui:Destroy()
    end

    -- Không xóa ATTRIBUTE_NAME:
    -- script không được kích hoạt lại trong phiên này.
end)
