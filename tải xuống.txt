--==================================================
-- NHAT HUB | SPEED ANTI-DROP
-- CLIENT ONLY | NO GUI
-- Giảm WalkSpeed >= 1: yêu cầu reset
-- Tăng WalkSpeed: không reset
--==================================================

local Players = game:GetService("Players")

local Player = Players.LocalPlayer
local DROP_THRESHOLD = 1

local SpeedConnection
local CharacterConnection
local CurrentHumanoid
local PreviousSpeed
local Resetting = false

-- Dọn kết nối theo dõi cũ nếu chạy lại script
if _G.NHATSpeedAntiDropCleanup then
    pcall(_G.NHATSpeedAntiDropCleanup)
end

local function StopWatching()
    if SpeedConnection then
        SpeedConnection:Disconnect()
        SpeedConnection = nil
    end
end

local function WatchCharacter(Character)
    StopWatching()

    Resetting = false
    PreviousSpeed = nil

    local Humanoid = Character:WaitForChild("Humanoid", 10)
    if not Humanoid or Player.Character ~= Character then
        return
    end

    CurrentHumanoid = Humanoid
    PreviousSpeed = Humanoid.WalkSpeed

    print("[NHAT HUB] Đang theo dõi WalkSpeed:", PreviousSpeed)

    SpeedConnection = Humanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
        if Resetting or Player.Character ~= Character then
            return
        end

        local OldSpeed = PreviousSpeed
        local NewSpeed = Humanoid.WalkSpeed

        -- Cập nhật mốc so sánh mỗi khi tốc độ thay đổi
        PreviousSpeed = NewSpeed

        -- Chỉ xử lý khi tốc độ giảm từ 1 đơn vị trở lên
        if OldSpeed and (OldSpeed - NewSpeed) >= DROP_THRESHOLD then
            Resetting = true

            warn(string.format(
                "[NHAT HUB] Phát hiện giảm WalkSpeed: %.3f -> %.3f",
                OldSpeed,
                NewSpeed
            ))

            -- Client yêu cầu nhân vật chết để kích hoạt
            -- cơ chế hồi sinh của game nếu máy chủ cho phép.
            pcall(function()
                if Player.Character == Character and Humanoid.Health > 0 then
                    Humanoid.Health = 0
                end
            end)
        end
    end)
end

local function Cleanup()
    StopWatching()

    if CharacterConnection then
        CharacterConnection:Disconnect()
        CharacterConnection = nil
    end
end

-- Lưu hàm dọn dẹp để tránh kết nối trùng khi chạy lại
_G.NHATSpeedAntiDropCleanup = Cleanup

-- Theo dõi nhân vật hiện tại
if Player.Character then
    task.spawn(WatchCharacter, Player.Character)
end

-- Theo dõi lại sau khi hồi sinh
CharacterConnection = Player.CharacterAdded:Connect(function(Character)
    task.wait(0.2)
    WatchCharacter(Character)
end)

print("[NHAT HUB] Speed Anti-Drop đã khởi động.")
